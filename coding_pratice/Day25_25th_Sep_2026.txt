# K8s self healing with Liveness & Rediness probes
apiversion: apps/v1
kind: Deployment
metadata:
 name: pathnex-web-app
spec:
 replicas: 3
 selector:
    matchLabels:
       app: pathnex-web-app
 template:
   metadata:
     labels:
       app: pathnex-web-app
spec:
 containers:
 -name: nginx
  image: nginx
  ports:
     - containerPort: 80
    liveness probe:
  httpGet:
   path: /healthz
   port:80
  initialdelaySeconds: 5
  periodSeconds: 10
 readinessProbe:
httpGet:
  path: /healthz
  port: 80
  initialDelaySeconds: 5
  periodSeconds: 10
